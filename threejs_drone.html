Vue3 / three.js 驱动片段
const gltf = await new GLTFLoader().loadAsync('/drone_mini3pro.glb');
scene.add(gltf.scene);

// 机头朝 -Z；旋翼在 glTF 里绕局部 Y 轴转
const rotors = ['FL','FR','RL','RR'].map(k =>
  gltf.scene.getObjectByName('Rotor_' + k));
const spin = [1, -1, -1, 1];   // 与 GLB extras 里的「转向」一致

function tick(dt, rpm) {
  const w = rpm * 2 * Math.PI / 60;
  rotors.forEach((r, i) => { r.rotation.y += spin[i] * w * dt; });
  gltf.scene.getObjectByName('Gimbal_Pivot')
       .rotation.x = THREE.MathUtils.degToRad(gimbalPitch);
}
